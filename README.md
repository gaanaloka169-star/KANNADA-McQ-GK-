  document.getElementById("editA").value = q.a;
  document.getElementById("editE").value = q.e;
  window.scrollTo({top: document.getElementById("admin").offsetTop, behavior:"smooth"});
}

function saveQuestion(){
  const q = document.getElementById("editQ").value.trim();
  const opts = [0,1,2,3].map(i=>document.getElementById("editO"+i).value.trim());
  const a = Number(document.getElementById("editA").value);
  const e = document.getElementById("editE").value.trim();

  if(!q || opts.some(x=>!x) || !e){
    alert("Question, 4 options ಮತ್ತು Explanation ಎಲ್ಲವೂ ತುಂಬಿ.");
    return;
  }

  const item = {q, o:opts, a, e};
  const index = Number(document.getElementById("editingIndex").value);

  if(index >= 0){
    questions[index] = item;
  }else{
    questions.push(item);
  }

  saveQuestions();
  renderQuiz();
  renderAdminQuestions();
  clearQuestionForm();
  alert(index >= 0 ? "Question updated!" : "New question added!");
}

function deleteQuestion(i){
  if(confirm("ಈ ಪ್ರಶ್ನೆಯನ್ನು delete ಮಾಡಬೇಕೇ?")){
    questions.splice(i,1);
    saveQuestions();
    renderQuiz();
    renderAdminQuestions();
  }
}

function renderAdminQuestions(){
  const box = document.getElementById("adminQuestionList");
  if(!box) return;
  box.innerHTML = "";
  if(!questions.length){
    box.innerHTML = "<p class='small'>No questions available.</p>";
    return;
  }
  questions.forEach((q,i)=>{
    const div = document.createElement("div");
    div.className = "admin-q";
    div.innerHTML =
      "<strong>"+(i+1)+". "+escapeHTML(q.q)+"</strong>" +
      "<div class='small'>A) "+escapeHTML(q.o[0])+" &nbsp; B) "+escapeHTML(q.o[1])+
      "<br>C) "+escapeHTML(q.o[2])+" &nbsp; D) "+escapeHTML(q.o[3])+
      "<br>Correct: "+String.fromCharCode(65+q.a)+"</div>" +
      "<div class='admin-actions'><button class='btn' onclick='editQuestion("+i+")'>✏️ Edit</button>"+
      "<button class='btn dark' onclick='deleteQuestion("+i+")'>🗑️ Delete</button></div>";
    box.appendChild(div);
  });
}

function escapeHTML(s){
  return String(s).replace(/[&<>"']/g, m=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#039;"}[m]));
}

function exportData(){
  const data = {siteSettings, questions};
  const blob = new Blob([JSON.stringify(data,null,2)], {type:"application/json"});
  const a = document.createElement("a");
  a.href = URL.createObjectURL(blob);
  a.download = "kannad-mcq-gk-backup.json";
  a.click();
  URL.revokeObjectURL(a.href);
}

function importData(event){
  const file = event.target.files[0];
  if(!file) return;
  const reader = new FileReader();
  reader.onload = function(){
    try{
      const data = JSON.parse(reader.result);
      if(Array.isArray(data.questions)) questions = data.questions;
      if(data.siteSettings) siteSettings = data.siteSettings;
      saveQuestions();
      saveSiteSettingsLocal();
      applySiteSettings();
      renderQuiz();
      renderAdminQuestions();
      loadAdminSettings();
      alert("Backup restored successfully!");
    }catch(err){
      alert("Invalid backup file.");
    }
  };
  reader.readAsText(file);
}

  renderAdminQuestions();

  clearQuestionForm();
  alert("Reset completed.");
}
